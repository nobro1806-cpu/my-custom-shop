const express = require('express');
const cors = require('cors');
const multer = require('multer');
const path = require('path');
const fs = require('fs');

const app = express();
app.use(cors());
app.use(express.json());

// Отдаем файлы из папки uploads
app.use('/uploads', express.static(path.join(__dirname, 'uploads')));

// Автоматически создаем папку uploads, если ее еще нет
if (!fs.existsSync('./uploads')) {
  fs.mkdirSync('./uploads');
}

// Настройка сохранения файлов
const storage = multer.diskStorage({
  destination: (req, file, cb) => cb(null, 'uploads/'),
  filename: (req, file, cb) => cb(null, Date.now() + path.extname(file.originalname))
});
const upload = multer({ storage });

// База данных в памяти (пока сервер запущен)
let products = [
  { id: 1, title: 'Игровая мышь Velix', price: 3500, img: 'https://via.placeholder.com/300' }
];

const ADMIN_PASSWORD = 'admin'; // Пароль твоей админки

// 1. Эндпоинт получения всех товаров
app.get('/api/products', (req, res) => {
  res.json(products);
});

// 2. Эндпоинт добавления товара с фото
app.post('/api/products', upload.single('image'), (req, res) => {
  const { title, price, password } = req.body;

  if (password !== ADMIN_PASSWORD) {
    return res.status(403).json({ error: 'Неверный пароль администратора!' });
  }

  if (!title || !price) {
    return res.status(400).json({ error: 'Заполните название и цену' });
  }

  const imgUrl = req.file 
    ? `${req.protocol}://${req.get('host')}/uploads/${req.file.filename}`
    : 'https://via.placeholder.com/300';

  const newProduct = {
    id: Date.now(),
    title,
    price: Number(price),
    img: imgUrl
  };

  products.push(newProduct);
  res.status(201).json(newProduct);
});

const PORT = 3000;
app.listen(PORT, () => console.log(`🚀 Сервер запущен на http://localhost:${PORT}`));
